25enum RequestId : uint8_t
26{
27 // No requests is from server, all requests are from Logic
28 // A reply whose value is null means "give up": the client declines to answer and the server
29 // applies the default reply (every legal reply value is non-null).
30 RequestInvalid = 0,
31
32 RequestRockPaperScissors, // request: array { string playerName } playerNames, int strivedOrder (or 0 for action) reply: int rps
33 RequestActionOrder, // request: array { int } remainedOrders, int maximumOrder, int selectionNum, reply: array { int } orders
34 RequestAction, // request: int currentOrder, reply: int(Action) action, optional string toPlayer, optional int toPlace
35 RequestUpgrade, // request: int remainingTimes, reply: array { int } item
36};
