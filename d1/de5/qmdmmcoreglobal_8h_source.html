158template<typename T>
159[[nodiscard]] QVariantList enumList2VariantList(const QList<T> &list)
160{
161 static_assert(std::is_enum_v<T>);
162
163 QVariantList ret;
164 ret.reserve(list.length());
165 foreach (T i, list)
166 ret << static_cast<int>(i);
167 return ret;
168}

170template<typename T>
171[[nodiscard]] QVariantList enumList2VariantList(const QList<QFlags<T>> &list)
172{
173 QVariantList ret;
174 ret.reserve(list.length());
175 foreach (const QFlags<T> &i, list)
176 ret << static_cast<int>(typename QFlags<T>::Int(i));
177 return ret;
178}
